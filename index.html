function buy(num, op) { const uPhone = localStorage.getItem('user'); const uName = localStorage.getItem('userName'); if(!uPhone) return openM('auth-m'); // 1. Сначала проверяем, не отправлял ли этот юзер заявку на ЭТОТ номер ранее db.ref('orders').orderByChild('wantedNum').equalTo(num).once('value', snap => { let alreadyExists = false; snap.forEach(orderSnap => { const order = orderSnap.val(); // Если номер совпал И телефон клиента совпал if(order.clientPhone === uPhone) { alreadyExists = true; } }); if(alreadyExists) { // Красивое предупреждение вместо стандартного alert const t = document.getElementById('toast'); t.querySelector('.gold-text').innerText = "Внимание"; t.querySelector('.font-bold').innerText = "Вы уже отправили заявку на этот номер. Ожидайте звонка."; showToast(); } else { // 2. Если такой заявки еще нет — создаем её db.ref('orders/' + Date.now()).set({ clientName: uName, clientPhone: uPhone, wantedNum: num, operator: op, date: new Date().toLocaleString(), status: 'new' }).then(() => { // Возвращаем стандартный текст уведомления const t = document.getElementById('toast'); t.querySelector('.gold-text').innerText = "Успешно"; t.querySelector('.font-bold').innerText = "Заявка в обработке, оператор свяжется с вами в ближайшее время."; showToast(); }); } }); }
